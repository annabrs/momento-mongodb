// NÍVEL 4: OPERAÇÕES E ESCRITÓRIOS
// -----------------------------------------------------------------------------

// 4.1 Qual é o custo total de suprimentos em cada escritório? Ordene do mais caro ao mais barato.
db.escritorios.aggregate([
  { $unwind: "$suprimentos" },
  {
    $group: {
      _id: "$nome",
      custoTotalSuprimentos: { $sum: { $multiply: ["$suprimentos.quantidade", "$suprimentos.preco_unitario"] } }
    }
  },
  { $sort: { custoTotalSuprimentos: -1 } }
]);

// 4.2 Qual escritório possui a maior quantidade de diferentes tipos de suprimentos?
db.escritorios.aggregate([
  { $project: { nome: 1, totalTiposSuprimentos: { $size: "$suprimentos" } } },
  { $sort: { totalTiposSuprimentos: -1 } },
  { $limit: 1 }
]);

// 4.3 Qual é o suprimento mais caro (considerando preço unitário) em toda a empresa?
db.escritorios.aggregate([
  { $unwind: "$suprimentos" },
  { $sort: { "suprimentos.preco_unitario": -1 } },
  { $limit: 1 },
  {
    $project: {
      escritorio: "$nome",
      suprimento: "$suprimentos.nome",
      preco_unitario: "$suprimentos.preco_unitario"
    }
  }
]);

// 4.4 Calcule o valor total do inventário de suprimentos da empresa.
db.escritorios.aggregate([
  { $unwind: "$suprimentos" },
  {
    $group: {
      _id: null,
      valorTotalInventario: { $sum: { $multiply: ["$suprimentos.quantidade", "$suprimentos.preco_unitario"] } }
    }
  }
]);

  { $group: { _id: "$cargo", total: { $sum: 1 } } },
  { $sort: { total: -1 } },
  { $limit: 3 }
]);

// 7.3 Funcionários que ganham acima da média salarial de seus respectivos departamentos.
db.funcionarios.aggregate([
  {
    $group: {
      _id: "$departamento",
      media_departamento: { $avg: "$salario" },
      funcionarios: { $push: "$$ROOT" }
    }
  },
  { $unwind: "$funcionarios" },
  {
    $match: {
      $expr: { $gt: ["$funcionarios.salario", "$media_departamento"] }
    }
  },
  { $replaceRoot: { newRoot: "$funcionarios" } }
]);

// 7.4 Calcule a taxa de crescimento da empresa por ano (contratações/ano).
db.funcionarios.aggregate([

  { $sort: { _id: 1 } }
]);

// NÍVEL 8: DESAFIOS E OTIMIZAÇÃO
// -----------------------------------------------------------------------------

// 8.1 Funcionários do departamento de Vendas com dependentes (cônjuge OU filhos).
db.funcionarios.find({
  departamento: "Vendas",
  $or: [
    { "conjuge": { $exists: true, $ne: null } },
    { "filhos": { $exists: true, $not: { $size: 0 } } }
  ]
});

// 8.2 Funcionários em escritórios diferentes do escritório do departamento.
db.funcionarios.aggregate([
  {
    $lookup: {
      from: "departamentos",
      localField: "departamento",
      foreignField: "nome",
      as: "dep_info"
    }
  },
  { $unwind: "$dep_info" },
  { $match: { $expr: { $ne: ["$escritorio", "$dep_info.escritorio"] } } }
]);

// 8.3 Relatório completo de um escritório.
db.escritorios.aggregate([
  {
    $lookup: {
      from: "departamentos",
      localField: "nome",
      foreignField: "escritorio",
      as: "departamentos"
    }
  },
  {
    $lookup: {
      from: "funcionarios",
      localField: "nome",
      foreignField: "escritorio",
      as: "funcionarios"
    }
  },
  {
    $addFields: {
      totalDepartamentos: { $size: "$departamentos" },
      totalFuncionarios: { $size: "$funcionarios" },
      custoSalarios: { $sum: "$funcionarios.salario" },
      custoSuprimentos: {
        $sum: {
          $map: {
            input: "$suprimentos",
            as: "sup",
            in: { $multiply: ["$$sup.quantidade", "$$sup.preco_unitario"] }
          }
        }
      }
    }
  },
  {
    $project: {
      nome: 1,
      pais: 1,
      totalDepartamentos: 1,
      totalFuncionarios: 1,
      custoSalarios: 1,
      custoSuprimentos: 1
    }
  }
]);

// 8.4 Encontrar o departamento mais equilibrado (menor diferença entre maior e menor salário).
db.funcionarios.aggregate([
  {
    $group: {
      _id: "$departamento",
      maiorSalario: { $max: "$salario" },
      menorSalario: { $min: "$salario" }
    }
  },
  { $addFields: { diferenca: { $subtract: ["$maiorSalario", "$menorSalario"] } } },
  { $sort: { diferenca: 1 } },
  { $limit: 1 }
]);

// 8.5 Text Search: Encontrar todos os produtos contendo "Uniforme".
db.produtos.find({ nome: { $regex: "Uniforme", $options: "i" } });

// 8.6 Date Range: Vendas no segundo trimestre de 2023 (abril a junho).
db.vendas.find({
  data_venda: {
    $gte: ISODate("2023-04-01T00:00:00Z"),
    $lte: ISODate("2023-06-30T23:59:59Z")
  }
});

// 9.1 Múltiplas Soluções - Salário entre $6.000 e $10.000:
// Forma A (Operadores de comparação):
db.funcionarios.find({ salario: { $gte: 6000, $lte: 10000 } });
// Forma B (Expressão de Agregação):
db.funcionarios.find({ $expr: { $and: [{ $gte: ["$salario", 6000] }, { $lte: ["$salario", 10000] }] } });
// Forma C (JavaScript / $where):
db.funcionarios.find({ $where: function() { return this.salario >= 6000 && this.salario <= 10000; } });

// 9.2 Otimização: A Opção A é a mais eficiente pois utiliza filtros nativos do BSON que podem aproveitar índices no MongoDB.
// 9.3 Data Quality: Funcionários sem e-mail ou telefone cadastrado.
db.funcionarios.find({
  $or: [
    { email: { $exists: false } }, { email: null }, { email: "" },
    { telefone: { $exists: false } }, { telefone: null }, { telefone: "" }
  ]
});

// 9.4 Funcionários solitários (únicos em seu cargo dentro de seu departamento).
db.funcionarios.aggregate([
  { $group: { _id: { departamento: "$departamento", cargo: "$cargo" }, total: { $sum: 1 }, doc: { $first: "$$ROOT" } } },
  { $match: { total: 1 } },
  { $replaceRoot: { newRoot: "$doc" } }
]);
